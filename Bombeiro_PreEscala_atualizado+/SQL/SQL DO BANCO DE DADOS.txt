CREATE DATABASE IF NOT EXISTS `escala_bombeiros`
  DEFAULT CHARACTER SET utf8mb4
  COLLATE utf8mb4_general_ci;

USE `escala_bombeiros`;

-- ==========================================
-- 1. ESTRUTURA DAS TABELAS
-- ==========================================

CREATE TABLE IF NOT EXISTS `bombeiro` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `nome` varchar(255) NOT NULL,
  `tipo` varchar(255) DEFAULT NULL,
  `email` varchar(255) NOT NULL,
  `senha_hash` varchar(255) NOT NULL,
  `perfil` varchar(255) DEFAULT NULL,
  `categoria_cnh` varchar(255) DEFAULT NULL,
  `habilidades` varchar(255) DEFAULT NULL,
  `curso_condutor_socorro` bit(1) DEFAULT NULL,
  `curso_socorrista` bit(1) DEFAULT NULL,
  `turma_12x36` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `email` (`email`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE IF NOT EXISTS `escala_plantao` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `data` date NOT NULL,
  `turno` varchar(255) DEFAULT NULL,
  `inicio` datetime DEFAULT NULL,
  `fim` datetime DEFAULT NULL,
  `status` varchar(50) NOT NULL DEFAULT 'ABERTA' COMMENT 'ABERTA ou FECHADA',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE IF NOT EXISTS `escala_clt_config` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `bombeiro_id` int(11) NOT NULL,
  `inicio_referencia` datetime NOT NULL,
  `ativo` tinyint(1) NOT NULL DEFAULT 1,
  PRIMARY KEY (`id`),
  UNIQUE KEY `bombeiro_id` (`bombeiro_id`),
  CONSTRAINT `fk_clt_config_bombeiro` FOREIGN KEY (`bombeiro_id`) REFERENCES `bombeiro` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE IF NOT EXISTS `escala_clt_fixa` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `bombeiro_id` int(11) NOT NULL,
  `inicio` datetime NOT NULL,
  `fim` datetime NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_clt_plantao` (`bombeiro_id`, `inicio`),
  CONSTRAINT `fk_clt_fixa_bombeiro` FOREIGN KEY (`bombeiro_id`) REFERENCES `bombeiro` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE IF NOT EXISTS `notificacao` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `escala_id` int(11) NOT NULL,
  `mensagem` varchar(255) DEFAULT NULL,
  `data_envio` datetime NOT NULL,
  `status` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `escala_id` (`escala_id`),
  CONSTRAINT `notificacao_ibfk_1` FOREIGN KEY (`escala_id`) REFERENCES `escala_plantao` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE IF NOT EXISTS `participacao` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `bombeiro_id` int(11) NOT NULL,
  `escala_id` int(11) NOT NULL,
  `status_presenca` tinyint(1) DEFAULT 0,
  `data_hora_cancelamento` datetime(6) DEFAULT NULL,
  `data_hora_reserva` datetime(6) DEFAULT NULL,
  `observacoes` varchar(255) DEFAULT NULL,
  `status_reserva` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `bombeiro_id` (`bombeiro_id`),
  KEY `escala_id` (`escala_id`),
  CONSTRAINT `participacao_ibfk_1` FOREIGN KEY (`bombeiro_id`) REFERENCES `bombeiro` (`id`) ON DELETE CASCADE,
  CONSTRAINT `participacao_ibfk_2` FOREIGN KEY (`escala_id`) REFERENCES `escala_plantao` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

CREATE TABLE IF NOT EXISTS `sessao` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `bombeiro_id` int(11) NOT NULL,
  `token` varchar(255) NOT NULL,
  `expira_em` datetime NOT NULL,
  PRIMARY KEY (`id`),
  KEY `bombeiro_id` (`bombeiro_id`),
  CONSTRAINT `sessao_ibfk_1` FOREIGN KEY (`bombeiro_id`) REFERENCES `bombeiro` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
